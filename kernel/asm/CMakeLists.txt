# JANUS asm — Inline Assembly Primitives
#
# Defines the 'janus_asm' INTERFACE target.
# The include path resolves #include <asm/foo.h> through the same
# include -> arch/include -> arch/<arch>/include chain used by the
# other kernel libraries.

add_library(janus_asm INTERFACE)

target_include_directories(janus_asm
    INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/arch/${JANUS_TARGET_ARCH}/include"
    "${PROJECT_SOURCE_DIR}/kernel/include"
)
janus_register(janus_asm ASM "")

# janus_asm is INTERFACE-only, so none of its headers ever appear in compile_commands.json on their own.
# We generate a throwaway Translation Unit to give clangd real, arch-correct compile commands for those headers.
file(GLOB JANUS_ASM_PUBLIC_HEADERS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/include/asm/*.h"
)

set(JANUS_ASM_CLANGD_DUMMY_CONTENT
"/* Generated by kernel/asm/CMakeLists.txt -- do not edit, and do not add to source control.
 * Throwaway TU giving clangd a real compile command for janus_asm's INTERFACE-only headers.
 * Never linked into the kernel image (EXCLUDE_FROM_ALL). */

")
foreach(_header ${JANUS_ASM_PUBLIC_HEADERS})
    get_filename_component(_header_name "${_header}" NAME)
    string(APPEND JANUS_ASM_CLANGD_DUMMY_CONTENT "#include <asm/${_header_name}>\n")
endforeach()

set(JANUS_ASM_CLANGD_DUMMY "${CMAKE_CURRENT_BINARY_DIR}/clangd_dummy.c")
file(WRITE "${JANUS_ASM_CLANGD_DUMMY}" "${JANUS_ASM_CLANGD_DUMMY_CONTENT}")

# EXCLUDE_FROM_ALL keeps this out of the default build
add_library(janus_asm_clangd_dummy OBJECT EXCLUDE_FROM_ALL "${JANUS_ASM_CLANGD_DUMMY}")
target_link_libraries(janus_asm_clangd_dummy PRIVATE janus_asm)
janus_apply_compile_flags(janus_asm_clangd_dummy)
